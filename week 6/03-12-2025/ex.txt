    <script>
      let add = document.getElementById("add");
      let task = document.getElementById("task");
      let data = document.getElementById("data");
      async function myFuncion(e) {
        let option = {
          method: "POST",
          headers: {
            "Content-type": "application/json",
          },
          body: JSON.stringify({
            task: `${task.value}`,
            status: "penting",
            flag: 0,
          }),
        };
        let res = fetch("http://localhost:3000/tasks", option);
      }
      add.addEventListener("click", myFuncion);
      async function main() {
        let res = await fetch("http://localhost:3000/tasks");
        let val = await res.json();
        for (let i in val) {
          if(`${val[i].task}`!=""){

            data.innerHTML += `<div class="sub">
                      <p onclick="doneFunction(event)" id=${val[i].id}>${val[i].task}</p>
                      <button onclick="editFunction(event)" data-id=${val[i].id} id="edit">edit</button>
                      <button onclick="delFunction(event)" data-id=${val[i].id} type="button">delete</button>
                      </div>
                      `;
          }
        }
      }
      main();
      async function delFunction(event) {
        let x = event.target.dataset.id;
        let res = await fetch("http://localhost:3000/tasks");
        let val = await res.json();
        for (let i in val) {
          if (val[i].id == x) {
            await fetch(`http://localhost:3000/tasks/${val[i].id}`, {
              method: "DELETE",
            });
          }
        }
      }
      async function editFunction(event) {
        let x = event.target.dataset.id;
        let b = document.getElementById("edit");
        let pop = document.getElementsByClassName("pop")[0];
        pop.classList.add("up");
        let popinp = document.getElementById("popinp");
        let edit2 = document.getElementById("edit2");
        let res = await fetch("http://localhost:3000/tasks");
        let val = await res.json();
        for (let i in val) {
          if (val[i].id == x) {
            popinp.value = `${val[i].task}`;
            edit2.addEventListener("click", async () => {
              pop.classList.remove("up");
              await fetch(`http://localhost:3000/tasks/${val[i].id}`, {
                method: "PATCH",
                headers: {
                  "Content-type": "application/json",
                },
                body: JSON.stringify({
                  task: `${popinp.value}`,
                }),
              });
            });
          }
        }
      }
      async function doneFunction(event) {
        let x = event.target;
        let c;
        let res = await fetch("http://localhost:3000/tasks");
        let val = await res.json();
        for (let i in val) {
          c = `${val[i].flag}`;
          if (val[i].task == x.innerText) {
            await fetch(`http://localhost:3000/tasks/${val[i].id}`, {
              method: "PATCH",
              headers: {
                "Content-type": "application/json",
              },
              body: JSON.stringify({
                status: "Done",
                flag: `${++c}`,
              }),
            });
          }
        }
      }
      async function cross() {
        let res = await fetch("http://localhost:3000/tasks");
        let val = await res.json();
        let para;
        for (let i in val) {
          if (val[i].status == "Done") {
            para = document.getElementById(`${val[i].id}`);
            para.innerHTML = `<del>${val[i].task}</del>`;
          }
        }
      }
      cross();

      async function nocross() {
        let res = await fetch("http://localhost:3000/tasks");
        let val = await res.json();
        let para;
        for (let i in val) {
          if (val[i].flag == 2) {
            para = document.getElementById(`${val[i].id}`);
            para.innerHTML = `${val[i].task}`;
            await fetch(`http://localhost:3000/tasks/${val[i].id}`, {
              method: "PATCH",
              headers: {
                "Content-type": "application/json",
              },
              body: JSON.stringify({
                status: "penting",
                flag: `${0}`,
              }),
            });
          }
        }
      }
      nocross();
